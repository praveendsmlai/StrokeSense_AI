SELECT
    CAST(due_date AS DATE) AS due_date,
    COUNT(*) AS total_actions,
    SUM(
        CASE
            WHEN owner_email IS NOT NULL
             AND LTRIM(RTRIM(owner_email)) <> ''
            THEN 1 ELSE 0
        END
    ) AS with_email,
    SUM(
        CASE
            WHEN owner_email IS NULL
              OR LTRIM(RTRIM(owner_email)) = ''
            THEN 1 ELSE 0
        END
    ) AS without_email
FROM dbo.pm_actions
WHERE status <> 'CLOSED'
  AND deleted_at IS NULL
  AND CAST(due_date AS DATE) BETWEEN
      '2026-10-24' AND '2026-10-26'
GROUP BY CAST(due_date AS DATE)
ORDER BY due_date;
