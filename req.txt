SELECT 'connect_insights' AS table_name,
       plan_connect_id AS connect_id
FROM dbo.connect_insights

UNION ALL

SELECT 'connect_minutes',
       plan_connect_id
FROM dbo.connect_minutes

UNION ALL

SELECT 'connect_actions',
       plan_connect_id
FROM dbo.connect_actions

UNION ALL

SELECT 'ai_generation_requests',
       plan_connect_id
FROM dbo.ai_generation_requests;
