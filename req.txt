import sqlite3
import uuid

c = sqlite3.connect("peopleconnect.db")
c.execute("PRAGMA foreign_keys=ON")

emp = c.execute(
    "SELECT employee_id, emp_name, email_id FROM employees WHERE employee_id=?",
    ("1025216",)
).fetchone()

assert emp is not None, "Employee not found"

user = c.execute(
    "SELECT id FROM application_users WHERE employee_id=?",
    (emp[0],)
).fetchone()

user_id = user[0] if user else str(uuid.uuid4())

try:
    if not user:
        c.execute(
            """INSERT INTO application_users
            (id, employee_id, email, display_name, is_active)
            VALUES (?, ?, ?, ?, ?)""",
            (user_id, emp[0], emp[2], emp[1], 1)
        )

    c.execute(
        """INSERT OR IGNORE INTO application_user_roles
        (user_id, role) VALUES (?, ?)""",
        (user_id, "PLATFORM_ADMIN")
    )

    c.commit()
    print("ADMIN SETUP SUCCESSFUL")

except Exception:
    c.rollback()
    raise

print(c.execute(
    """SELECT u.employee_id, u.display_name, r.role
    FROM application_users u
    JOIN application_user_roles r ON r.user_id = u.id
    WHERE u.employee_id = ?""",
    ("1025216",)
).fetchall())

c.close()
