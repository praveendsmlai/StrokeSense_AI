from sqlalchemy import create_engine
from sqlalchemy.orm import Session

from tests.integration.arun_schema.models import ArunTestBase
from tests.integration.arun_schema.fixture import _seed

engine = create_engine("sqlite:///peopleconnect.db")

with engine.begin() as connection:
    existing = set(
        __import__("sqlalchemy").inspect(connection).get_table_names()
    )
    arun_tables = set(ArunTestBase.metadata.tables.keys())

    if existing & arun_tables:
        raise RuntimeError("Arun tables already exist. Setup stopped.")

    ArunTestBase.metadata.create_all(connection)

try:
    with Session(engine) as session:
        _seed(session)
        session.commit()
except Exception:
    with engine.begin() as connection:
        ArunTestBase.metadata.drop_all(connection)
    raise

print("ARUN DUMMY SETUP SUCCESSFUL")
