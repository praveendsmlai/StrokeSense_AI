"""Screenshot-faithful provisional Arun lifecycle models for Phase 2B tests.

Source: ``ARUN_LIFECYCLE_SCHEMA_SPEC.md`` §§4–5.  These are test-only
transcriptions, not verified Arun DDL; their metadata is never production metadata.
"""
from __future__ import annotations

from datetime import date, datetime, time

from sqlalchemy import Boolean, CheckConstraint, Date, DateTime, ForeignKey, Integer, String, Text, Time, UniqueConstraint
from sqlalchemy.orm import DeclarativeBase, Mapped, mapped_column, relationship

CONNECT_STATUSES = ("DRAFT", "INITIATED", "IN_PROGRESS", "SCHEDULED", "COMPLETED", "CANCELLED")
CONNECT_STAGES = ("INITIATE", "REVIEW", "EVIDENCE_CAPTURE", "COMPLETED")
WORKFLOW_TYPES = ("PLANNED", "BACKDATED")
MEETING_MODES = ("TEAMS", "HYBRID", "IN_PERSON")
PARTICIPANT_SOURCES = ("SEARCH", "UPLOAD")
PARTICIPANT_STATUSES = ("VALID", "INACTIVE", "DUPLICATE", "UNRESOLVED", "OUT_OF_SCOPE", "INVALID")
UPLOAD_STATUSES = ("UPLOADED", "PROCESSING", "VALIDATED", "PARTIAL_SUCCESS", "FAILED")
CONNECT_TYPE_STATUSES = ("ACTIVE", "INACTIVE", "RETIRED")
USER_ROLES = ("EMPLOYEE", "BHR", "BHR_LEADER", "PLATFORM_ADMIN")
ENRICHMENT_STATUSES = ("SUCCESS", "FAILED")


class ArunTestBase(DeclarativeBase):
    """Dedicated test metadata; names may coincide with production tables."""


class ArunUser(ArunTestBase):
    __tablename__ = "users"
    __table_args__ = (CheckConstraint(f"role IN {USER_ROLES}", name="ck_test_users_role"),)
    id: Mapped[str] = mapped_column(String(36), primary_key=True)
    email: Mapped[str] = mapped_column(String(255), nullable=False, unique=True, index=True)
    name: Mapped[str] = mapped_column(String(255), nullable=False)
    employee_id: Mapped[str] = mapped_column(String(64), nullable=False, unique=True, index=True)
    department: Mapped[str] = mapped_column(String(255), nullable=False, index=True)
    role: Mapped[str] = mapped_column(String(32), nullable=False, default="BHR", index=True)
    is_active: Mapped[bool] = mapped_column(Boolean, nullable=False, default=True, index=True)
    created_at: Mapped[datetime] = mapped_column(DateTime, nullable=False)
    updated_at: Mapped[datetime] = mapped_column(DateTime, nullable=False)
    initiations = relationship("ArunPlanConnectInitiation", back_populates="initiator")
    uploads = relationship("ArunPlanConnectFileUpload", back_populates="uploader")

class ArunPlanConnectInitiation(ArunTestBase):
    __tablename__ = "plan_connect_initiations"
    __table_args__ = (
        CheckConstraint(f"status IN {CONNECT_STATUSES}", name="ck_test_initiations_status"),
        CheckConstraint(f"current_stage IN {CONNECT_STAGES}", name="ck_test_initiations_stage"),
        CheckConstraint(f"workflow_type IN {WORKFLOW_TYPES}", name="ck_test_initiations_workflow"),
        CheckConstraint(f"meeting_mode IN {MEETING_MODES}", name="ck_test_initiations_mode"),
    )
    id: Mapped[str] = mapped_column(String(64), primary_key=True)
    initiator_id: Mapped[str] = mapped_column(ForeignKey("users.id"), nullable=False, index=True)
    client_request_id: Mapped[str | None] = mapped_column(String(128), nullable=True, index=True)
    connect_type_id: Mapped[str | None] = mapped_column(ForeignKey("connect_types.connect_type_id"), nullable=True, index=True)
    connect_type_name: Mapped[str | None] = mapped_column(String(150), nullable=True)
    subject: Mapped[str | None] = mapped_column(String(255), nullable=True)
    bhr_firstname: Mapped[str | None] = mapped_column(String(128), nullable=True)
    bhr_lastname: Mapped[str | None] = mapped_column(String(128), nullable=True)
    bhr_email: Mapped[str | None] = mapped_column(String(255), nullable=True)
    bhr_employee_ibu: Mapped[str | None] = mapped_column(String(128), nullable=True)
    meeting_date: Mapped[date | None] = mapped_column(Date, nullable=True)
    start_time: Mapped[time | None] = mapped_column(Time, nullable=True)
    end_time: Mapped[time | None] = mapped_column(Time, nullable=True)
    scheduled_from: Mapped[datetime | None] = mapped_column(DateTime, nullable=True)
    scheduled_to: Mapped[datetime | None] = mapped_column(DateTime, nullable=True)
    meeting_mode: Mapped[str | None] = mapped_column(String(16), nullable=True)
    additional_emails: Mapped[str | None] = mapped_column(Text, nullable=True)
    agenda: Mapped[str | None] = mapped_column(Text, nullable=True)
    workflow_type: Mapped[str | None] = mapped_column(String(16), nullable=True, index=True)
    status: Mapped[str] = mapped_column(String(16), nullable=False, default="DRAFT", index=True)
    current_stage: Mapped[str] = mapped_column(String(32), nullable=False, index=True)
    participant_count: Mapped[int] = mapped_column(Integer, nullable=False, default=0)
    teams_deeplink: Mapped[str | None] = mapped_column(Text, nullable=True)
    confirmed_accuracy: Mapped[bool] = mapped_column(Boolean, nullable=False, default=False)
    follow_up_days: Mapped[int | None] = mapped_column(Integer, nullable=True)
    invited_count: Mapped[int] = mapped_column(Integer, nullable=False, default=0)
    emails_sent_at: Mapped[datetime | None] = mapped_column(DateTime, nullable=True)
    sent_by: Mapped[str | None] = mapped_column(String(64), nullable=True)
    row_version: Mapped[int] = mapped_column(Integer, nullable=False, default=1)
    created_at: Mapped[datetime] = mapped_column(DateTime, nullable=False, index=True)
    updated_at: Mapped[datetime] = mapped_column(DateTime, nullable=False)
    deleted_at: Mapped[datetime | None] = mapped_column(DateTime, nullable=True)
    initiator = relationship("ArunUser", back_populates="initiations")
    connect_type = relationship("ArunConnectType", back_populates="initiations")
    participants = relationship("ArunPlanConnectParticipant", back_populates="initiation")
    uploads = relationship("ArunPlanConnectFileUpload", back_populates="initiation")
