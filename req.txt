development database URL.

class UnsafeLifecycleFixtureTarget(ValueError):
    """Raised when a disposable fixture could target a non-temporary database."""

def assert_isolated_database_path(database_path: Path, *, allowed_root: Path) -> Path:
    """Return a resolved disposable SQLite path or reject it before any I/O."""
    target = database_path.resolve()
    root = allowed_root.resolve()
    try:
        target.relative_to(root)
    except ValueError as exc:
        raise UnsafeLifecycleFixtureTarget("fixture database must be inside its temporary root") from exc
    if target.name.lower() == "peopleconnect.db":
        raise UnsafeLifecycleFixtureTarget("fixture database cannot be named peopleconnect.db")
    if target.suffix.lower() not in {".db", ".sqlite", ".sqlite3"}:
        raise UnsafeLifecycleFixtureTarget("fixture database must be a SQLite database file")
    return target


def sqlite_url_for(database_path: Path) -> str:
    return f"sqlite:///{database_path.resolve().as_posix()}"


def assert_isolated_database_url(url: str, *, allowed_root: Path) -> Path:
    """Reject non-SQLite, in-memory, and non-temporary URLs before connection."""
    parsed = urlparse(url)
    if parsed.scheme != "sqlite" or not parsed.path or parsed.path in {":memory:", "/:memory:"}:
        raise UnsafeLifecycleFixtureTarget("fixture requires a file-backed SQLite URL")
    # SQLAlchemy turns a Windows absolute path into /C:/... in a sqlite URL.
    return assert_isolated_database_path(Path(unquote(parsed.path.lstrip("/"))), allowed_root=allowed_root)


@dataclass
class ArunFixtureDatabase:
    path: Path
    allowed_root: Path
    engine: Engine
    session_factory: sessionmaker[Session]

    def session(self) -> Session:
        return self.session_factory()

    def dispose(self) -> None:
        self.engine.dispose()
        target = assert_isolated_database_path(self.path, allowed_root=self.allowed_root)
        if target.exists():
            target.unlink()


def create_seeded_fixture_database(database_path: Path, *, allowed_root: Path) -> ArunFixtureDatabase:
    """Create a new disposable database and seed every documented lifecycle case."""
    target = assert_isolated_database_path(database_path, allowed_root=allowed_root)
    if target.exists():
        raise UnsafeLifecycleFixtureTarget("fixture target already exists; refusing to overwrite it")
    target.parent.mkdir(parents=True, exist_ok=True)
    url = sqlite_url_for(target)
    assert_isolated_database_url(url, allowed_root=allowed_root)
    engine = create_engine(url)

    @event.listens_for(engine, "connect")
    def _enable_foreign_keys(dbapi_connection, _connection_record) -> None:
        cursor = dbapi_connection.cursor()
        cursor.execute("PRAGMA foreign_keys=ON")
        cursor.close()

    ArunTestBase.metadata.create_all(engine)
    fixture = ArunFixtureDatabase(
        path=target,
        allowed_root=allowed_root.resolve(),
        engine=engine,
        session_factory=sessionmaker(bind=engine),
    )
    session = fixture.session()
    try:
        _seed(session)
        session.commit()
    except Exception:
        session.rollback()
        fixture.dispose()
        raise
    finally:
        session.close()
    return fixture


def fixture_table_names(engine: Engine) -> frozenset[str]:
    return frozenset(inspect(engine).get_table_names())

    meeting_date = date(2026, 11, 1) if workflow_type else None

    end = time(10, 30) if workflow_type else None
    return ArunPlanConnectInitiation(
        id=connect_id, initiator_id=_USER_BHR, client_request_id=f"fixture-{connect_id}",
        connect_type_id=connect_type.connect_type_id, connect_type_name=connect_type.name,

        bhr_email="bhr.fixture@example.test", bhr_employee_ibu="IBU-A", meeting_date=meeting_date,
        start_time=start, end_time=end,
        scheduled_from=datetime(2026, 11, 1, 10, 0) if workflow_type else None,
        scheduled_to=datetime(2026, 11, 1, 10, 30) if workflow_type else None,
        meeting_mode=meeting_mode, additional_emails="[]", agenda="Synthetic agenda", workflow_type=workflow_type,
        status=status, current_stage=stage, participant_count=participant_count,
