import sqlite3

conn = sqlite3.connect("peopleconnect.db")

tables = [
    "plan_connect_initiations",
    "plan_connect_participants",
    "participant_snapshots",
    "plan_connect_file_uploads",
    "participant_upload_rows",
    "connect_types",
    "hr_roster",
    "users",
    "audit_logs",
]

for table in tables:
    exists = conn.execute(
        "SELECT name FROM sqlite_master WHERE type='table' AND name=?",
        (table,)
    ).fetchone()

    if exists:
        count = conn.execute(
            f'SELECT COUNT(*) FROM "{table}"'
        ).fetchone()[0]
        print(f"{table}: EXISTS | Records: {count}")
    else:
        print(f"{table}: NOT FOUND")

conn.close()
